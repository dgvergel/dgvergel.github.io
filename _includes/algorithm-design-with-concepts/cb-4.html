template<    forward_iterator I,    sentinel_for<I> S,    typename Comp = ranges::less,    typename Proj = identity > requires sortable<I, Comp, Proj> // ----------------------------- constexpr void insertion_sort(I first, S last, Comp comp = Comp{}, Proj proj = Proj{}) {    // return immediately if the range length is <= 1:    if constexpr (sized_sentinel_for<S, I>) {       if (last - first <= 1) return; // (A)    } else {       if (first == last or ranges::next(first) == last) return; // (B)    }    for (I const i : views::iota(ranges::next(first), last)) {       I const j = ranges::upper_bound(first, i, invoke(proj, *i), comp, proj); // (C)       ranges::rotate(j, i, ranges::next(i)); // (D)    } }
