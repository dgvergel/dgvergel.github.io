struct Book {    string title;    double price; }; auto books = vector<Book>{    {.title = "Romeo and Juliet", .price = 8.95},    { "The Count of Monte Cristo", 12.89},    { "War and Peace", 7.59},    { "Chronicle of a Death Foretold", 10.59},    { "A Study in Scarlet", 9.89} }; insertion_sort(books, ranges::greater{}, &Book::price); for (Book const& b : books) {    println("title: {:<30} | price: {} €", b.title, b.price); } /* output:    title: The Count of Monte Cristo | price: 12.89 €    title: Chronicle of a Death Foretold | price: 10.59 €    title: A Study in Scarlet | price: 9.89 €    title: Romeo and Juliet | price: 8.95 €    title: War and Peace | price: 7.59 € */
