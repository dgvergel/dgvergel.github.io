template<plain_aggregate T> [[nodiscard]] auto parse_row(    std::string_view csv_line,    char delimiter = ',' )    -> std::expected<T, parse_error> {    detail::assert_valid_members<T>();    using enum parse_error_code;    auto fields = csv_line | std::views::split(delimiter);    auto field  = fields.begin();    auto res = T{};    auto column = 1uz;    template for (constexpr std::meta::info dm : data_members<T>) {       if (field == fields.end()) {          return std::unexpected{parse_error{missing_value, column}};       }       auto& q = res.[: dm :];       if (          auto r = detail::parse_value(             std::string_view{*field}, q.numerical_value_ref_in(q.unit)          );          not r       ) {          return std::unexpected{parse_error{r.error(), column}};       }       ++field;       ++column;    }    if (field != fields.end()) {       return std::unexpected{parse_error{extra_value, column}};    }    return res; }
