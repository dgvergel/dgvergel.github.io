namespace detail { template<typename T> concept charconv_parsable =    requires (char const* first, char const* last, T& value) {       { std::from_chars(first, last, value) } -> std::same_as<std::from_chars_result>;    }; template<typename T> concept stream_extractable =    requires (std::istream& is, T& value) {       { is >> value } -> std::same_as<std::istream&>;    }; template<typename T> concept parsable_value = charconv_parsable<T> or stream_extractable<T>;
