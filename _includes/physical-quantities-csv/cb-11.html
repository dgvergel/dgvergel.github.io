struct csv_options {    char delimiter = ',';    bool has_header = true; }; template<detail::plain_aggregate T> [[nodiscard]] auto load_quantities(    std::istream& is,    csv_options opts = {} )    -> std::expected<std::vector<T>, load_error> {    auto res = std::vector<T>{};    auto buffer = std::string{};    auto line = 1uz;    if (opts.has_header and std::getline(is, buffer)) {       ++line;    }    while (std::getline(is, buffer)) {       auto parsed = detail::parse_row<T>(buffer, opts.delimiter);       if (not parsed) {          return std::unexpected{load_error{row_error{line, parsed.error()}}};       }       res.push_back(std::move(*parsed));       ++line;    }    if (is.bad()) {       return std::unexpected{load_error{read_error{line}}};    }    return res; } template<detail::plain_aggregate T> [[nodiscard]] auto load_quantities(    std::filesystem::path const& path,    csv_options opts = {} )    -> std::expected<std::vector<T>, file_load_error> {    auto file = std::ifstream{path};    if (not file) {       return std::unexpected{file_load_error{open_file_error{}}};    }    return load_quantities<T>(file, opts)       .transform_error([](load_error const& e) {          return e.visit([](auto const& alt) { return file_load_error{alt}; });       }); } } // namespace qcsv
