inline constexpr auto ascii_whitespace = std::string_view{" \t\n\r\f\v"}; template<parsable_value T> [[nodiscard]] auto parse_value(std::string_view token, T& value)    -> std::expected<void, parse_error_code> {    using enum parse_error_code;    auto const fnw = token.find_first_not_of(ascii_whitespace);    if (fnw == std::string_view::npos) {       return std::unexpected{missing_value};    }    token = token.substr(fnw, token.find_last_not_of(ascii_whitespace) - fnw + 1);    if constexpr (charconv_parsable<T>) {       auto const last = token.data() + token.size();       auto const [p, ec] = std::from_chars(token.data(), last, value);       if (ec != std::errc{} or p != last) {          return std::unexpected{format_error};       }    }    else {       auto input = std::ispanstream{token};       input.imbue(std::locale::classic());       if (not (input >> value) or input.peek() != std::char_traits<char>::eof()) {          return std::unexpected{format_error};       }    }    return {}; }
