#include <cstdlib> #include <print> #include "quantity_csv.hpp" #include <mp-units/systems/isq_angle.h> #include <mp-units/systems/si.h> #include <mp-units/systems/yard_pound.h> namespace mpu = mp_units; using namespace mpu::si::unit_symbols; using namespace mpu::yard_pound::unit_symbols; using angle_type =    mpu::quantity<       mpu::isq_angle::angular_measure[          mpu::angular::unit_symbols::deg]>; using speed_type =    mpu::quantity<mpu::isq::speed[mi/h]>; using range_type =    mpu::quantity<mpu::isq::distance[cm]>; struct shot_data {    angle_type phi;    speed_type v;    range_type r; }; auto main() -> int {    auto const data =       qcsv::load_quantities<shot_data>("parabolic_throwing_data.csv");    if (not data) {       std::println(stderr, "CSV error: {}", data.error());       return EXIT_FAILURE;    }    for (auto&& [phi, v, r] : *data) {       std::println(          "{::N[.1f]}, {::N[.1f]}, {::N[.1f]}", phi, v, r);    }    return EXIT_SUCCESS; }
